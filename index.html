Actual Sales = 
SUMIFS (
    Sales[SalesAmount],

    Sales[Date], ">="&[@Month],

    AND ( [@[Achievement%]] >= 100%, [@[Attendance%]] >= 90% ),

// 3. Tiered Commission Slab Lookup (No Nested IFs)

    CommissionSlab[Rate],
    0,

sql/07_procedures_triggers.sql — Stored Procedure sp_monthly_mis_report 
CREATE PROCEDURE sp_monthly_mis_report(IN p_month CHAR(7))
BEGIN
    DECLARE v_prev CHAR(7);
    SET v_prev = DATE_FORMAT(STR_TO_DATE(CONCAT(p_month, '-01'), '%Y-%m-%d')
                             - INTERVAL 1 MONTH, '%Y-%m');

    -- 1. KPI Scorecard: This Month vs Previous Month
    SELECT
        k.kpi,
        k.this_month,
        k.prev_month,
        ROUND(100 * (k.this_month - k.prev_month) / NULLIF(k.prev_month, 0), 2) AS change_pct
    FROM (
        SELECT 'Orders' AS kpi,
               SUM(purchase_month = p_month) AS this_month,
               SUM(purchase_month = v_prev)  AS prev_month
        FROM mart_order_summary WHERE is_valid_order = 1
        UNION ALL
        SELECT 'GMV (BRL)',
               ROUND(SUM(CASE WHEN purchase_month = p_month THEN gmv END), 2),
               ROUND(SUM(CASE WHEN purchase_month = v_prev  THEN gmv END), 2)
        FROM mart_order_summary WHERE is_valid_order = 1
    ) k;
END$$

Materialized Mart & High-Performance Views

To eliminate expensive multi-table joins during ad-hoc BI exploration in Power BI or Excel, built a materialized mart_order_summary populated via automated ETL.

  • Single-Order Grain: Collapses order items, payments, customer demographics, and latest reviews into a unified 98K-row fact table.
  • Window Function Analytics: vw_monthly_kpi uses LAG(gmv) OVER (ORDER BY purchase_month) to compute dynamic month-over-month growth rates and YTD cumulative revenues.
  • Automated Refresh: Built sp_refresh_order_mart() to truncate and reload the materialized mart in a scheduled ETL job.
sql/05_mis_views.sql — Executive View with LAG() vw_monthly_kpi 
CREATE OR REPLACE VIEW vw_monthly_kpi AS
WITH m AS (
    SELECT purchase_month,
           COUNT(*)                                       AS orders,
           COUNT(DISTINCT customer_unique_id)             AS active_customers,
           ROUND(SUM(product_revenue), 2)                 AS product_revenue,
           ROUND(SUM(gmv), 2)                             AS gmv,
           ROUND(AVG(gmv), 2)                             AS avg_order_value,
           ROUND(100 * AVG(is_late), 2)                   AS late_delivery_pct,
           ROUND(AVG(delivery_days), 1)                   AS avg_delivery_days,
           ROUND(AVG(review_score), 2)                    AS avg_review_score
    FROM mart_order_summary
    WHERE is_valid_order = 1
      AND purchase_month BETWEEN '2017-01' AND '2018-08'
    GROUP BY purchase_month
)
SELECT m.*,
       -- Month-over-Month GMV Growth % via LAG Window Function
       ROUND(100 * (gmv - LAG(gmv) OVER (ORDER BY purchase_month))
                 / LAG(gmv) OVER (ORDER BY purchase_month), 2)   AS gmv_mom_growth_pct,
       ROUND(SUM(gmv) OVER (PARTITION BY LEFT(purchase_month, 4)
                            ORDER BY purchase_month), 2)        AS gmv_ytd
FROM m;
